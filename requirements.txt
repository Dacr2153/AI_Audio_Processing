# Audio Restoration — Requirements
# =================================
# This project uses pyproject.toml as the single source of truth for
# dependencies (PEP 621). Do not add packages here; add them to
# pyproject.toml under [project.dependencies] or [project.optional-dependencies].
#
# Install the package with its dependencies:
#   pip install -e .
# Include neural models (torch, deepfilternet, demucs):
#   pip install -e ".[neural]"
# Also install dev tooling (pytest, ruff, mypy, build):
#   pip install -e ".[dev]"
# AudioSR super-resolution (requires Python <= 3.10):
#   pip install -e ".[audiosr]"
#
# Or use the automated (idempotent) setup script:
#   bash setup.sh            # CPU build
#   bash setup.sh --gpu      # CUDA build
#   bash setup.sh --dev      # + dev tools
